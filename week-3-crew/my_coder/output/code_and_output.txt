```python
# Leibniz formula for π approximation: π/4 = 1 - 1/3 + 1/5 - 1/7 + ...
# Multiply sum by 4 to get π approximation.
# Calculate first 10,000 terms.

def leibniz_pi(n):
    total = 0.0
    sign = 1          # starts positive for first term (k=0)
    for k in range(n):
        total += sign / (2 * k + 1)
        sign = -sign  # alternate sign each term
    return total * 4

n_terms = 10000
approx_pi = leibniz_pi(n_terms)
print(f"Approximation of π using {n_terms} terms: {approx_pi}")
```

**Output:**
```
Approximation of π using 10000 terms: 3.141392653591791
```
